import os
import base64
from fastapi import APIRouter, Request, HTTPException
from groq import Groq

router = APIRouter()

@router.post("/troubleshoot")
@router.post("/troubleshoot-vision")
async def troubleshoot_chat(request: Request):
    api_key = os.getenv("GROQ_API_KEY")
    if not api_key:
        raise HTTPException(status_code=500, detail="GROQ_API_KEY is missing in Render environment variables.")

    client = Groq(api_key=api_key)
    content_type = request.headers.get("content-type", "")

    message = ""
    base64_image = None
    farm_context = ""

    if "application/json" in content_type:
        body = await request.json()
        message = body.get("message") or body.get("query") or body.get("prompt") or ""
        base64_image = body.get("image")
        farm_context = body.get("context") or ""
    else:
        form = await request.form()
        message = form.get("message") or form.get("query") or "Analyze this agricultural query."
        farm_context = form.get("context") or ""
        upload_file = form.get("file") or form.get("image")
        if upload_file and hasattr(upload_file, "read"):
            file_bytes = await upload_file.read()
            if file_bytes:
                base64_image = base64.b64encode(file_bytes).decode("utf-8")

    if not message and not base64_image:
        raise HTTPException(status_code=400, detail="Please provide a question or an image.")

    try:
        if base64_image:
            if "," in base64_image:
                base64_image = base64_image.split(",")[1]
            completion = client.chat.completions.create(
                model="llama-3.2-11b-vision-preview",
                messages=[
                    {
                        "role": "user",
                        "content": [
                            {
                                "type": "text",
                                "text": f"You are an expert Indian plant pathologist and agronomist. Diagnose this image and answer directly with exact remedies and dosages: {message}"
                            },
                            {
                                "type": "image_url",
                                "image_url": {"url": f"data:image/jpeg;base64,{base64_image}"}
                            }
                        ]
                    }
                ],
                temperature=0.4,
                max_tokens=900
            )
        else:
            system_prompt = (
                "You are AgriBrain, an elite AI Agronomist for Indian farming. "
                "Give direct, practical, scientific answers immediately with exact numerical ranges "
                "(temperature in °C, NPK in kg/ha, soil pH, seed rate, irrigation schedule, and Indian market/MSP insights). "
                "NEVER respond with a generic template asking the user for their crop or location."
            )
            if farm_context:
                system_prompt += f"\nActive Farm Digital Twin Context: {farm_context}"

            completion = client.chat.completions.create(
                model="llama-3.3-70b-versatile",
                messages=[
                    {"role": "system", "content": system_prompt},
                    {"role": "user", "content": message}
                ],
                temperature=0.5,
                max_tokens=900
            )

        reply = completion.choices[0].message.content
        return {"response": reply, "reply": reply, "message": reply, "status": "success"}

    except Exception as e:
        raise HTTPException(status_code=500, detail=f"Groq AI Error: {str(e)}")

"use client";
import React, { useEffect, useMemo, useRef } from "react";
import { MapContainer, TileLayer, Marker, Popup, useMapEvents, useMap } from "react-leaflet";
import L from "leaflet";
import "leaflet/dist/leaflet.css";

const svgPin = L.divIcon({
  className: "custom-pin-icon",
  html: `<div style="transform: translate(-16px, -40px); cursor: grab;">
    <svg width="36" height="44" viewBox="0 0 24 30" fill="none" xmlns="http://www.w3.org/2000/svg">
      <path d="M12 0C5.37 0 0 5.37 0 12C0 21 12 30 12 30C12 30 24 21 24 12C24 5.37 18.63 0 12 0Z" fill="#10B981" stroke="#ffffff" stroke-width="1.5"/>
      <circle cx="12" cy="11" r="5" fill="#0F172A"/>
    </svg>
  </div>`,
  iconSize: [36, 44],
  iconAnchor: [18, 44],
});

interface MapUIProps {
  pin: { lat: number; lng: number } | null;
  onPinDrop: (lat: number, lng: number) => void;
}

function ClickHandler({ onPinDrop }: { onPinDrop: (lat: number, lng: number) => void }) {
  useMapEvents({
    click(e) {
      onPinDrop(e.latlng.lat, e.latlng.lng);
    },
  });
  return null;
}

function MapFlyTo({ pin }: { pin: { lat: number; lng: number } | null }) {
  const map = useMap();
  useEffect(() => {
    if (pin) {
      map.flyTo([pin.lat, pin.lng], Math.max(map.getZoom(), 7), { duration: 0.75 });
    }
  }, [pin, map]);
  return null;
}

export default function MapUI({ pin, onPinDrop }: MapUIProps) {
  const markerRef = useRef<any>(null);
  const defaultCenter: [number, number] = [20.5937, 78.9629];
  const markerPos: [number, number] | null = pin ? [pin.lat, pin.lng] : null;

  const eventHandlers = useMemo(
    () => ({
      dragend() {
        const marker = markerRef.current;
        if (marker != null) {
          const pos = marker.getLatLng();
          onPinDrop(pos.lat, pos.lng);
        }
      },
    }),
    [onPinDrop]
  );

  return (
    <div className="w-full h-[320px] sm:h-[390px] lg:h-[430px] rounded-2xl overflow-hidden border border-slate-700/80 shadow-2xl relative z-0">
      <MapContainer center={defaultCenter} className="w-full h-full z-0" scrollWheelZoom={true} zoom={5}>
        <TileLayer attribution="&copy; <a href=\"https://www.openstreetmap.org/copyright\">OpenStreetMap</a>"
          url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
        />
        <ClickHandler onPinDrop={onPinDrop}/>
        <MapFlyTo pin={pin}/>
        {markerPos && (
          <Marker draggable={true} eventHandlers={eventHandlers} icon={svgPin} position={markerPos} ref={markerRef}>
            <Popup>
              <div className="text-xs font-semibold text-slate-900">
                📍 Farm Pin: {pin?.lat.toFixed(4)}°N, {pin?.lng.toFixed(4)}°E
                <br />
                <span className="text-emerald-700">(Drag pin or click map to adjust)</span>
              </div>
            </Popup>
          </Marker>
        )}
      </MapContainer>
    </div>
  );
}
