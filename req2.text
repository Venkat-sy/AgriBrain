                <div className="grid grid-cols-2 gap-3">
                  <button
                    type="button"
                    onClick={() => setSetupType("short-term")}
                    className={`py-3 px-4 rounded-xl font-medium text-sm transition-all border ${
                      setupType === "short-term"
                        ? "bg-emerald-600 text-white border-emerald-400 shadow-lg shadow-emerald-600/20"
                        : "bg-slate-800/80 text-slate-300 border-slate-700 hover:bg-slate-800"
                    }`}
                  >
                    🌾 Short-Term / Seasonal
                  </button>
                  <button
                    type="button"
                    onClick={() => setSetupType("long-term")}
                    className={`py-3 px-4 rounded-xl font-medium text-sm transition-all border ${
                      setupType === "long-term"
                        ? "bg-emerald-600 text-white border-emerald-400 shadow-lg shadow-emerald-600/20"
                        : "bg-slate-800/80 text-slate-300 border-slate-700 hover:bg-slate-800"
                    }`}
                  >
                    🌴 Long-Term / Orchard
                  </button>
                </div>
              </div>

              <button
                type="button"
                onClick={runSimulation}
                disabled={loading}
                className="w-full py-3.5 px-6 rounded-xl font-semibold text-white bg-gradient-to-r from-emerald-500 to-teal-600 hover:from-emerald-400 hover:to-teal-500 disabled:opacity-50 shadow-lg shadow-emerald-600/25 transition-all"
              >
                {loading ? "⚡ Running XGBoost + SHAP Digital Twin..." : "⚡ Run Full Digital Twin Simulation"}
              </button>
            </div>
          </div>

          {/* Right Column: Always-Visible Telemetry & Results Cards (5 cols on Desktop) */}
          <div className="lg:col-span-5 space-y-4">
            {/* Live Weather & Soil Telemetry Grid */}
            <div className="grid grid-cols-2 gap-4">
              <div className={`bg-slate-900/90 border border-slate-800 rounded-2xl p-4 shadow-xl ${loading ? "animate-pulse" : ""}`}>
                <span className="text-xs font-semibold uppercase tracking-wider text-sky-400">
                  ☀️ Live Weather
                </span>
                <div className="mt-3 space-y-2 text-sm">
                  <div className="flex justify-between">
                    <span className="text-slate-400">Temp:</span>
                    <span className="font-semibold text-white">{data?.weather?.temp ?? "--"} °C</span>
                  </div>
                  <div className="flex justify-between">
                    <span className="text-slate-400">Humidity:</span>
                    <span className="font-semibold text-white">{data?.weather?.humidity ?? "--"} %</span>
                  </div>
                  <div className="flex justify-between">
                    <span className="text-slate-400">Rainfall:</span>
                    <span className="font-semibold text-white">{data?.weather?.rainfall ?? "--"} mm</span>
                  </div>
                </div>
              </div>

              <div className={`bg-slate-900/90 border border-slate-800 rounded-2xl p-4 shadow-xl ${loading ? "animate-pulse" : ""}`}>
                <span className="text-xs font-semibold uppercase tracking-wider text-amber-400">
                  🌱 Soil NPK & pH
                </span>
                <div className="mt-3 space-y-1.5 text-sm">
                  <div className="flex justify-between">
                    <span className="text-slate-400">Nitrogen (N):</span>
                    <span className="font-semibold text-white">{data?.soil?.N ?? "--"} kg/ha</span>
                  </div>
                  <div className="flex justify-between">
                    <span className="text-slate-400">Phosphorus (P):</span>
                    <span className="font-semibold text-white">{data?.soil?.P ?? "--"} kg/ha</span>
                  </div>
                  <div className="flex justify-between">
                    <span className="text-slate-400">Potassium (K):</span>
                    <span className="font-semibold text-white">{data?.soil?.K ?? "--"} kg/ha</span>
                  </div>
                  <div className="flex justify-between">
                    <span className="text-slate-400">Soil pH:</span>
                    <span className="font-semibold text-emerald-400">{data?.soil?.pH ?? "--"}</span>
                  </div>
                </div>
              </div>
            </div>

            {/* ML Recommendation & SHAP Card */}
            <div className={`bg-slate-900/90 border border-slate-800 rounded-2xl p-5 shadow-xl space-y-3 ${loading ? "animate-pulse" : ""}`}>
              <div className="flex items-center justify-between">
                <span className="text-xs font-semibold uppercase tracking-wider text-emerald-400">
                  🤖 AI Recommended Crop
                </span>
                <span className="text-xs px-2.5 py-1 rounded-full bg-emerald-500/15 text-emerald-300 font-semibold">
                  Confidence: {data?.confidence ? `${Math.round(data.confidence * 100)}%` : "--%"}
                </span>
              </div>

              <div className="text-2xl font-bold text-white">
                {data?.recommended_crop || "Awaiting Simulation Run"}
              </div>

              <p className="text-xs text-slate-300 leading-relaxed bg-slate-800/70 p-3 rounded-xl border border-slate-700/60">
                <strong className="text-emerald-400">SHAP Explainability: </strong>
                {data?.shap_data?.reasoning || "Drop a pin on the map and click 'Run Full Digital Twin Simulation' to compute SHAP feature importance."}
              </p>

              {data?.alternatives && data.alternatives.length > 0 && (
                <div className="pt-2">
                  <span className="text-xs text-slate-400 block mb-1.5">Top Ranked Alternatives:</span>
                  <div className="flex flex-wrap gap-2">
                    {data.alternatives.map((alt: any, idx: number) => (
                      <span key={idx} className="text-xs px-2.5 py-1 rounded-lg bg-slate-800 text-slate-200 border border-slate-700">
                        {alt.crop} ({alt.score_pct}%)
                      </span>
                    ))}
                  </div>
                </div>
              )}
            </div>

            {/* Cultivation Timeline Card */}
            <div className={`bg-slate-900/90 border border-slate-800 rounded-2xl p-5 shadow-xl space-y-3 ${loading ? "animate-pulse" : ""}`}>
              <span className="text-xs font-semibold uppercase tracking-wider text-purple-400 block">
                📅 AI Cultivation Stage Timeline
              </span>
              {data?.timeline && data.timeline.length > 0 ? (
                <div className="space-y-3 max-h-[300px] overflow-y-auto pr-1">
                  {data.timeline.map((stage: any, i: number) => (
                    <div key={i} className="p-3 rounded-xl bg-slate-800/60 border border-slate-700/60 space-y-1">
                      <div className="flex justify-between items-center text-sm font-semibold text-white">
                        <span>{i + 1}. {stage.stage_name}</span>
                        <span className="text-xs text-emerald-400">{stage.days || stage.estimated_days} Days</span>
                      </div>
                      <p className="text-xs text-slate-300 leading-relaxed">{stage.agronomy_advice}</p>
                    </div>
                  ))}
                </div>
              ) : (
                <p className="text-xs text-slate-400 py-4 text-center">
                  Stage-by-stage agronomy schedule will populate here immediately after running the simulation.
                </p>
              )}
            </div>
          </div>
        </div>
      </div>
    </div>
  );
}
